CREATE TABLE Users (
    user_id NUMBER GENERATED ALWAYS AS IDENTITY,
    username VARCHAR2(50) NOT NULL,
    password_hash VARCHAR2(255) NOT NULL,
    email VARCHAR2(100) NOT NULL,
    phone VARCHAR2(20) NOT NULL,
    role VARCHAR2(10) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    deleted_at TIMESTAMP NULL,
    CONSTRAINT USERS_PK PRIMARY KEY (user_id),
    CONSTRAINT USERS_CK_ROLE CHECK (role IN ('TEACHER', 'PARENT'))
);


CREATE UNIQUE INDEX USERS_UQ_USERNAME ON Users (CASE WHEN deleted_at IS NULL THEN username END);
CREATE UNIQUE INDEX USERS_UQ_EMAIL ON Users (CASE WHEN deleted_at IS NULL THEN email END);
CREATE UNIQUE INDEX USERS_UQ_PHONE ON Users (CASE WHEN deleted_at IS NULL THEN phone END);


CREATE TABLE Teachers (
    teacher_id NUMBER GENERATED ALWAYS AS IDENTITY,
    user_id NUMBER NOT NULL,
    teacher_code VARCHAR2(20) NOT NULL,
    full_name VARCHAR2(100) NOT NULL,
    contact_info CLOB,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    deleted_at TIMESTAMP NULL,
    CONSTRAINT TEACHERS_PK PRIMARY KEY (teacher_id),
    CONSTRAINT TEACHERS_FK_USER FOREIGN KEY (user_id) REFERENCES Users (user_id) ON DELETE CASCADE
);


CREATE UNIQUE INDEX TEACHERS_UQ_USER_ID ON Teachers (CASE WHEN deleted_at IS NULL THEN user_id END);
CREATE UNIQUE INDEX TEACHERS_UQ_CODE ON Teachers (CASE WHEN deleted_at IS NULL THEN teacher_code END);


CREATE TABLE Parents (
    parent_id NUMBER GENERATED ALWAYS AS IDENTITY,
    user_id NUMBER NOT NULL,
    full_name VARCHAR2(100) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    deleted_at TIMESTAMP NULL,
    CONSTRAINT PARENTS_PK PRIMARY KEY (parent_id),
    CONSTRAINT PARENTS_FK_USER FOREIGN KEY (user_id) REFERENCES Users (user_id) ON DELETE CASCADE
);


CREATE UNIQUE INDEX PARENTS_UQ_USER_ID ON Parents (CASE WHEN deleted_at IS NULL THEN user_id END);


CREATE TABLE Classrooms (
    class_id NUMBER GENERATED ALWAYS AS IDENTITY,
    class_name VARCHAR2(50) NOT NULL,
    class_code_6digit VARCHAR2(6) NOT NULL,
    academic_year NUMBER(4) NOT NULL,
    semester NUMBER(1) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    deleted_at TIMESTAMP NULL,
    CONSTRAINT CLASSROOMS_PK PRIMARY KEY (class_id),
    CONSTRAINT CLASSROOMS_CK_SEMESTER CHECK (semester IN (1, 2, 3))
);


CREATE UNIQUE INDEX CLASSROOMS_UQ_CODE ON Classrooms (CASE WHEN deleted_at IS NULL THEN class_code_6digit END);


CREATE TABLE Students (
    student_id NUMBER GENERATED ALWAYS AS IDENTITY,
    student_code VARCHAR2(20) NOT NULL,
    student_name VARCHAR2(100) NOT NULL,
    status VARCHAR2(20) DEFAULT 'ACTIVE' NOT NULL,
    parent_id NUMBER NOT NULL,
    class_id NUMBER NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT STUDENTS_PK PRIMARY KEY (student_id),
    CONSTRAINT STUDENTS_FK_PARENT FOREIGN KEY (parent_id) REFERENCES Parents (parent_id),
    CONSTRAINT STUDENTS_FK_CLASSROOM FOREIGN KEY (class_id) REFERENCES Classrooms (class_id),
    CONSTRAINT STUDENTS_CK_STATUS CHECK (status IN ('ACTIVE', 'GRADUATED', 'SUSPENDED'))
);


CREATE UNIQUE INDEX STUDENTS_UQ_CODE ON Students (student_code);


CREATE TABLE Teacher_Class (
    teacher_id NUMBER NOT NULL,
    class_id NUMBER NOT NULL,
    CONSTRAINT TEACHER_CLASS_PK PRIMARY KEY (teacher_id, class_id),
    CONSTRAINT TEACHER_CLASS_FK_TEACHER FOREIGN KEY (teacher_id) REFERENCES Teachers (teacher_id) ON DELETE CASCADE,
    CONSTRAINT TEACHER_CLASS_FK_CLASSROOM FOREIGN KEY (class_id) REFERENCES Classrooms (class_id) ON DELETE CASCADE
);


CREATE TABLE Homework (
    homework_id NUMBER GENERATED ALWAYS AS IDENTITY,
    teacher_id NUMBER NOT NULL,
    class_id NUMBER NOT NULL,
    title VARCHAR2(150) NOT NULL,
    description CLOB,
    file_path VARCHAR2(255),
    link_url VARCHAR2(255),
    due_date TIMESTAMP NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT HOMEWORK_PK PRIMARY KEY (homework_id),
    CONSTRAINT HOMEWORK_FK_TEACHER FOREIGN KEY (teacher_id) REFERENCES Teachers (teacher_id),
    CONSTRAINT HOMEWORK_FK_CLASSROOM FOREIGN KEY (class_id) REFERENCES Classrooms (class_id) ON DELETE CASCADE
);


CREATE TABLE HomeworkSubmission (
    submission_id NUMBER GENERATED ALWAYS AS IDENTITY,
    homework_id NUMBER NOT NULL,
    student_id NUMBER NOT NULL,
    teacher_id NUMBER NULL,
    submiss_file VARCHAR2(255),
    submiss_url VARCHAR2(255),
    score NUMBER(5, 2),
    feedback CLOB,
    submiss_status VARCHAR2(20) DEFAULT 'PENDING' NOT NULL,
    submitted_at TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT SUBMISSION_PK PRIMARY KEY (submission_id),
    CONSTRAINT SUBMISSION_FK_HOMEWORK FOREIGN KEY (homework_id) REFERENCES Homework (homework_id) ON DELETE CASCADE,
    CONSTRAINT SUBMISSION_FK_STUDENT FOREIGN KEY (student_id) REFERENCES Students (student_id) ON DELETE CASCADE,
    CONSTRAINT SUBMISSION_FK_TEACHER FOREIGN KEY (teacher_id) REFERENCES Teachers (teacher_id) ON DELETE SET NULL,
    CONSTRAINT SUBMISSION_CK_SCORE CHECK (score IS NULL OR score >= 0),
    CONSTRAINT SUBMISSION_CK_STATUS CHECK (submiss_status IN ('PENDING', 'SUBMITTED', 'GRADED', 'LATE'))
);


CREATE TABLE NewsPost (
    news_id NUMBER GENERATED ALWAYS AS IDENTITY,
    teacher_id NUMBER NOT NULL,
    title VARCHAR2(200) NOT NULL,
    content CLOB NOT NULL,
    post_date TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    is_pinned NUMBER(1) DEFAULT 0 NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    deleted_at TIMESTAMP NULL,
    CONSTRAINT NEWS_POST_PK PRIMARY KEY (news_id),
    CONSTRAINT NEWS_POST_FK_TEACHER FOREIGN KEY (teacher_id) REFERENCES Teachers (teacher_id),
    CONSTRAINT NEWS_POST_CK_PINNED CHECK (is_pinned IN (0, 1))
);



CREATE TABLE PostImage (
    image_id NUMBER GENERATED ALWAYS AS IDENTITY,
    news_id NUMBER NOT NULL,
    image_url VARCHAR2(500) NOT NULL,
    caption VARCHAR2(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT POST_IMAGE_PK PRIMARY KEY (image_id),
    CONSTRAINT POST_IMAGE_FK_NEWS FOREIGN KEY (news_id) REFERENCES NewsPost (news_id) ON DELETE CASCADE
);


CREATE TABLE WeeklyMenu (
    menu_id NUMBER GENERATED ALWAYS AS IDENTITY,
    teacher_id NUMBER NOT NULL,
    week_number NUMBER(2) NOT NULL,
    week_start_date DATE NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT WEEKLY_MENU_PK PRIMARY KEY (menu_id),
    CONSTRAINT WEEKLY_MENU_FK_TEACHER FOREIGN KEY (teacher_id) REFERENCES Teachers (teacher_id),
    CONSTRAINT WEEKLY_MENU_CK_WEEK CHECK (week_number BETWEEN 1 AND 53)
);



CREATE TABLE MenuItem (
    item_id NUMBER GENERATED ALWAYS AS IDENTITY,
    menu_id NUMBER NOT NULL,
    day_of_week VARCHAR2(3) NOT NULL,
    food_name VARCHAR2(150) NOT NULL,
    CONSTRAINT MENU_ITEM_PK PRIMARY KEY (item_id),
    CONSTRAINT MENU_ITEM_FK_MENU FOREIGN KEY (menu_id) REFERENCES WeeklyMenu (menu_id) ON DELETE CASCADE,
    CONSTRAINT MENU_ITEM_CK_DAY CHECK (day_of_week IN ('MON', 'TUE', 'WED', 'THU', 'FRI'))
);


CREATE TABLE Attendance (
    att_id NUMBER GENERATED ALWAYS AS IDENTITY,
    student_id NUMBER NOT NULL,
    class_id NUMBER NOT NULL,
    teacher_id NUMBER NOT NULL,
    att_date DATE NOT NULL,
    att_status VARCHAR2(10) NOT NULL,
    remark VARCHAR2(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT ATTENDANCE_PK PRIMARY KEY (att_id),
    CONSTRAINT ATTENDANCE_FK_STUDENT FOREIGN KEY (student_id) REFERENCES Students (student_id) ON DELETE CASCADE,
    CONSTRAINT ATTENDANCE_FK_CLASSROOM FOREIGN KEY (class_id) REFERENCES Classrooms (class_id) ON DELETE CASCADE,
    CONSTRAINT ATTENDANCE_FK_TEACHER FOREIGN KEY (teacher_id) REFERENCES Teachers (teacher_id),
    CONSTRAINT ATTENDANCE_CK_STATUS CHECK (att_status IN ('PRESENT', 'LATE', 'ABSENT', 'LEAVE'))
);



CREATE TABLE DailyRecord (
    record_id NUMBER GENERATED ALWAYS AS IDENTITY,
    student_id NUMBER NOT NULL,
    class_id NUMBER NOT NULL,
    teacher_id NUMBER NOT NULL,
    record_date DATE NOT NULL,
    health_status CLOB,
    behavior_review CLOB,
    pdf_report_path VARCHAR2(255),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    CONSTRAINT DAILY_RECORD_PK PRIMARY KEY (record_id),
    CONSTRAINT DAILY_RECORD_FK_STUDENT FOREIGN KEY (student_id) REFERENCES Students (student_id) ON DELETE CASCADE,
    CONSTRAINT DAILY_RECORD_FK_CLASSROOM FOREIGN KEY (class_id) REFERENCES Classrooms (class_id) ON DELETE CASCADE,
    CONSTRAINT DAILY_RECORD_FK_TEACHER FOREIGN KEY (teacher_id) REFERENCES Teachers (teacher_id)
);


-- 1.1 สำหรับตาราง Attendance (ค้นหาประวัติการเช็กชื่อตามห้องเรียนและวันที่)
CREATE INDEX idx_att_class_date ON Attendance(class_id, att_date);

-- 1.2 สำหรับตาราง DailyRecord (ค้นหาบันทึกประจำวันตามนักเรียนและวันที่)
CREATE INDEX idx_record_student_date ON DailyRecord(student_id, record_date);

-- 1.3 สำหรับตาราง Homework (ค้นหาการบ้านตามห้องเรียนและวันกำหนดส่ง)
CREATE INDEX idx_homework_class_due ON Homework(class_id, due_date);


-- 2.1 FK Indexes สำหรับ student_id
CREATE INDEX idx_fk_students_parent ON Students(parent_id);
CREATE INDEX idx_fk_students_class ON Students(class_id);
CREATE INDEX idx_fk_submission_student ON HomeworkSubmission(student_id);

-- 2.2 FK Indexes สำหรับ teacher_id
CREATE INDEX idx_fk_homework_teacher ON Homework(teacher_id);
CREATE INDEX idx_fk_submission_teacher ON HomeworkSubmission(teacher_id);
CREATE INDEX idx_fk_att_teacher ON Attendance(teacher_id);
CREATE INDEX idx_fk_record_teacher ON DailyRecord(teacher_id);
CREATE INDEX idx_fk_news_teacher ON NewsPost(teacher_id);
CREATE INDEX idx_fk_menu_teacher ON WeeklyMenu(teacher_id);

-- 2.3 FK Indexes สำหรับ parent_id / user_id
CREATE INDEX idx_fk_teacher_user ON Teachers(user_id);
CREATE INDEX idx_fk_parent_user ON Parents(user_id);
